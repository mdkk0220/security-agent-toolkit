```
def score(count, country, threshold, points):
    total = 0
    if country != "KR":
        total += 30
    if count >= threshold:
        total += points
    return total

def decide(total):
    if total >= 50:
        return "거부"
    if total >= 20:
        return "추가확인"
    return "허용"

for name, threshold, points in PLANS:
    false_positive = 0
    false_negative = 0
    for r in RECORDS:
        d = decide(score(r["count"], r["country"], threshold, points))
        if r["kind"] == "정상" and d != "허용":
            false_positive += 1
        if r["kind"] == "위험" and d != "거부":
            false_negative += 1
    print(name, "오탐", false_positive, "미탐", false_negative)
```
